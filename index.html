ENDPOINT PUBLIK, BERISIKO. // Deploy: supabase functions deploy replace-energy --no-verify-jwt import { createClient } from 'npm:@supabase/supabase-js@2'; const cors={'Access-Control-Allow-Origin':'https://cikobebjn.github.io','Access-Control-Allow-Headers':'apikey, content-type','Access-Control-Allow-Methods':'POST, OPTIONS'}; Deno.serve(async(req)=>{ if(req.method==='OPTIONS')return new Response(null,{status:204,headers:cors}); const send=(data,status=200)=>new Response(JSON.stringify(data),{status,headers:{...cors,'content-type':'application/json'}}); if(req.method!=='POST')return send({error:'Method not allowed'},405); const origin=req.headers.get('origin'); if(origin!=='https://cikobebjn.github.io')return send({error:'Origin tidak diizinkan'},403); try{ const payload=await req.json();const rows=payload?.rows; if(!Array.isArray(rows)||rows.length<1||rows.length>2000)return send({error:'Jumlah data harus 1-2000 baris'},400); const serviceKey=Deno.env.get('SUPABASE_SERVICE_ROLE_KEY'); const url=Deno.env.get('SUPABASE_URL'); if(!serviceKey||!url)return send({error:'Konfigurasi server belum lengkap'},503); const db=createClient(url,serviceKey,{auth:{persistSession:false}}); const {data,error}=await db.rpc('replace_energy_daily',{p_rows:rows}); if(error)throw Error(error.message); return send(data||{row_count:rows.length}); }catch(e){return send({error:String(e?.message||e)},400);} });
